// --- CONFIG & DATASETS ---
const GLOBAL_WHITELIST = [
    "google.com", "facebook.com", "youtube.com", "netflix.com",
    "microsoft.com", "apple.com", "linkedin.com", "instagram.com",
    "twitter.com", "wikipedia.org", "github.com", "canvas.instructure.com",
    "zoom.us"
];

const TARGETS = ['paypal.com', 'facebook.com', 'bdo.com.ph', 'gcash.com'];

// --- LAYER 1: NORMALIZATION & DEBOUNCE ---
let lastClickTime = 0;
document.addEventListener('click', (e) => {
    const link = e.target.closest('a');
    if (!link) return;

    const now = Date.now();
    if (now - lastClickTime < 500) return; 
    lastClickTime = now;

    if (link.href.startsWith('http')) {
        e.preventDefault(); 
        processURL(link.href);
    }
});

function normalizeURL(url) {
    try {
        let clean = url.replace(/^(https?:\/\/)?(www\.)?/, '').split(/[?#]/)[0];
        return clean.toLowerCase().replace(/\/$/, "");
    } catch (e) { return url; }
}

// --- LAYER 2: DETERMINISTIC FILTERING (D1) ---
async function processURL(rawUrl) {
    const timerId = "DetectionLatency_" + Date.now();
    console.time(timerId);
    
    const startTime = performance.now();
    const normalized = normalizeURL(rawUrl);

    chrome.storage.session.get(['scanCount'], (res) => {
        chrome.storage.session.set({ scanCount: (res.scanCount || 0) + 1 });
    });

    chrome.storage.local.get(['whitelist', 'blocklist'], (result) => {
        const userWhitelist = result.whitelist || [];
        const blocklist = result.blocklist || [];

        if (GLOBAL_WHITELIST.some(domain => normalized.includes(domain))) {
            window.location.href = rawUrl; 
            console.timeEnd(timerId);
            return;
        }

        if (userWhitelist.some(domain => normalized.includes(domain))) {
            window.location.href = rawUrl; 
            console.timeEnd(timerId);
            return;
        }

        if (blocklist.some(domain => normalized.includes(domain))) {
            showNotification(`🚨 ${normalized} is on your Blocklist.`);
            console.timeEnd(timerId);
            return;
        }

        runInference(rawUrl, normalized, startTime, timerId);
    });
}

function runInference(rawUrl, normalized, startTime, timerId) {
    const len = normalized.length;
    const dots = (normalized.match(/\./g) || []).length;
    const entropy = [...new Set(normalized)].reduce((acc, char) => {
        const p = (normalized.split(char).length - 1) / len;
        return acc - p * Math.log2(p);
    }, 0);
    
    const minDistance = Math.min(...TARGETS.map(t => levenshtein(normalized, t)));
    const score = (len * 0.01) + (dots * 0.15) + (entropy * 0.1) + (1 / (minDistance + 1) * 0.6);
    const probability = Math.min(Math.round(score * 100), 99);

    const endTime = performance.now();
    const latency = (endTime - startTime).toFixed(2);
    console.timeEnd(timerId);

    showXAINarrative(rawUrl, normalized, probability, latency);
}

function levenshtein(s, t) {
    if (!s.length) return t.length; if (!t.length) return s.length;
    const arr = [];
    for (let i = 0; i <= t.length; i++) { arr[i] = [i]; }
    for (let j = 0; j <= s.length; j++) { arr[0][j] = j; }
    for (let i = 1; i <= t.length; i++) {
        for (let j = 1; j <= s.length; j++) {
            arr[i][j] = t[i - 1] === s[j - 1] ? arr[i - 1][j - 1] : Math.min(arr[i - 1][j - 1] + 1, arr[i][j - 1] + 1, arr[i - 1][j] + 1);
        }
    }
    return arr[t.length][s.length];
}

function showNotification(msg) {
    const banner = document.createElement('div');
    banner.innerText = msg;
    banner.style.cssText = `
        position: fixed; top: 20px; left: 50%; transform: translateX(-50%);
        background: #333; color: white; padding: 12px 24px; border-radius: 8px;
        z-index: 2147483647; font-family: sans-serif; font-size: 14px;
        box-shadow: 0 10px 20px rgba(0,0,0,0.3); opacity: 0; transition: opacity 0.3s;
    `;
    document.body.appendChild(banner);
    setTimeout(() => banner.style.opacity = '1', 10);
    setTimeout(() => {
        banner.style.opacity = '0';
        setTimeout(() => banner.remove(), 300);
    }, 3000);
}

function addToList(type, url) {
    chrome.storage.local.get([type], (result) => {
        let list = result[type] || [];
        if (!list.includes(url)) {
            list.unshift(url); 
            let update = {};
            update[type] = list;
            chrome.storage.local.set(update, () => {
                showNotification(`Added to ${type.charAt(0).toUpperCase() + type.slice(1)}`);
            });
        }
    });
}
